struct StateEntry { char name[80]{};uint64_t bytes=0,hash=0; };
struct Q4State {
 std::vector<StateEntry> values;std::vector<uint8_t> buffer;uint64_t read_bytes=0;
 void add(const std::string& name,const void* src,uint64_t n,bool gpu){if(!src||!n)return;StateEntry e{};require_name(name);std::memcpy(e.name,name.c_str(),name.size());e.bytes=n;uint64_t h=1469598103934665603ull;buffer.resize(4*1024*1024);cudaPointerAttributes attr{};if(gpu){auto err=cudaPointerGetAttributes(&attr,src);Q4Tape::require(err==cudaSuccess,"state pointer attributes");if(attr.type==cudaMemoryTypeHost){src=attr.hostPointer;gpu=false;}}
  auto* p=(const uint8_t*)src;for(uint64_t off=0;off<n;){size_t take=std::min<uint64_t>(buffer.size(),n-off);const uint8_t* b=p+off;if(gpu){Q4Tape::require(cudaMemcpy(buffer.data(),b,take,cudaMemcpyDeviceToHost)==cudaSuccess,"state D2H read");b=buffer.data();}for(size_t i=0;i<take;++i){h^=b[i];h*=1099511628211ull;}off+=take;}e.hash=h;read_bytes+=n;values.push_back(e);
 }
 static void require_name(const std::string& n){Q4Tape::require(n.size()<80,"state component name");}
 void qsa(const std::string& prefix,const strata::core::QsaState& q,int64_t used){auto a=[&](const char* n,const void* p,uint64_t bytes){add(prefix+"."+n,p,bytes,true);};int64_t shape[]={q.max_cells,q.n_pages,q.n_slots,q.kv_mode,q.idx_pooled_rows,q.kv_int8,q.kv_rot};add(prefix+".shape",shape,sizeof(shape),false);
  a("page_table",q.page_table,q.n_pages*4);if(q.kv_mode==1){a("slot_block",q.map.slot_block,q.map.n_slots*4);a("slot_stamp",q.map.slot_stamp,q.map.n_slots*4);a("slot_ref",q.map.slot_ref,q.map.n_slots*4);a("map_ctl",q.map.ctl,strata::kernels::kKvCtlInts*4);}
  a("idx_tail",q.idx_tail,3*128*4);a("idx_dead",q.idx_dead,128*4);a("idx_pooled",q.idx_pooled,q.idx_pooled_rows*128*4);a("idx_block_pos",q.idx_block_pos,4);
  uint64_t slots=q.n_slots,logical=std::min<int64_t>(q.n_pages,used/4);if(q.kv_mode!=1){a("k_q",q.k_q,slots*4*2*256);a("v_q",q.v_q,slots*4*2*256);a("k_scale",q.k_scale,slots*4*2*4*2);a("v_scale",q.v_scale,slots*4*2*4*2);} // Streamed pools are redundant authoritative-host copies; maps checked, canonical host payload below.
  if(q.host.present()&&q.kv_mode==1){a("host_k_q",q.host.k_q,logical*4*2*256);a("host_v_q",q.host.v_q,logical*4*2*256);a("host_k_scale",q.host.k_scale,logical*4*2*4*2);a("host_v_scale",q.host.v_scale,logical*4*2*4*2);int rem=(int)(used%4);if(rem)for(int h=0;h<2;++h){auto tag="tail"+std::to_string(h);a(("host_k_q."+tag).c_str(),q.host.k_q+logical*2048+h*1024,rem*256);a(("host_v_q."+tag).c_str(),q.host.v_q+logical*2048+h*1024,rem*256);a(("host_k_scale."+tag).c_str(),q.host.k_scale+logical*32+h*16,rem*4*2);a(("host_v_scale."+tag).c_str(),q.host.v_scale+logical*32+h*16,rem*4*2);}}
 }
 void session(const std::string& prefix,strata::core::SessionState& s,const strata::core::ModelGeometry& g,int64_t used){add(prefix+".GDN",s.gdn_state,s.gdn_alloc*((uint64_t)g.ssm_state_size*g.ssm_v_heads*g.ssm_state_size+(uint64_t)g.ssm_conv_channels*(g.ssm_d_conv-1))*4,true);add(prefix+".HC_R",s.R,g.hc_dim()*4,true);add(prefix+".PLE_hist",s.ple_hist,strata::kernels::NG_HIST*strata::kernels::NG_HC_DIM*4,true);add(prefix+".PLE_prev",s.ple_prev,sizeof(s.ple_prev),false);add(prefix+".PLE_token",&s.ple_token,4,false);for(int i=0;i<s.qsa_alloc;++i)qsa(prefix+".QSA"+std::to_string(s.qsa_ord0+i),s.qsa_states[s.qsa_ord0+i],used);}
 void begin(strata::core::SessionState& s0,strata::core::SessionState& s1,const strata::core::QsaState& mtp,const strata::core::ModelGeometry& g,int64_t used){if(!q4_tape.active)return;uint64_t start=Q4Tape::ns();values.clear();read_bytes=0;int old;cudaGetDevice(&old);cudaSetDevice(0);Q4Tape::require(cudaDeviceSynchronize()==cudaSuccess,"state stage0 sync");session("GPU0",s0,g,used-1);cudaSetDevice(1);Q4Tape::require(cudaDeviceSynchronize()==cudaSuccess,"state stage1 sync");session("GPU1",s1,g,used-1);qsa("MTP",mtp,used);cudaSetDevice(old);auto path=q4_tape.path+".initial-state.bin";
  if(q4_tape.replaying){FILE* f=std::fopen(path.c_str(),"rb");Q4Tape::require(f,"initial-state sidecar missing");uint64_t n=0;Q4Tape::io(f,&n,1,false);Q4Tape::require(n==values.size(),"initial-state component count");std::vector<StateEntry> expected(n);Q4Tape::io(f,expected.data(),n,false);Q4Tape::require(std::fgetc(f)==EOF,"initial-state sidecar trailing");std::fclose(f);int mismatches=0;for(size_t i=0;i<n;++i)if(std::memcmp(&expected[i],&values[i],sizeof(StateEntry))){++mismatches;std::fprintf(stderr,"Q4_STATE_MISMATCH component=%s expected=%016llx actual=%016llx bytes=%llu\n",values[i].name,(unsigned long long)expected[i].hash,(unsigned long long)values[i].hash,(unsigned long long)values[i].bytes);}Q4Tape::require(!mismatches,"initial KV/GDN/PLE/MTP state differs");}
  else{FILE* f=std::fopen(path.c_str(),"wbx");Q4Tape::require(f,"state record path exists");uint64_t n=values.size();Q4Tape::io(f,&n,1,true);Q4Tape::io(f,values.data(),n,true);std::fclose(f);}
  const char* p=std::getenv("STRATA_Q4_TAPE_OBSERVATIONS");if(p){auto name=std::string(p)+".initial-state.bin";FILE* f=std::fopen(name.c_str(),"wbx");Q4Tape::require(f,"state observation exists");uint64_t n=values.size();Q4Tape::io(f,&n,1,true);Q4Tape::io(f,values.data(),n,true);std::fclose(f);}std::fprintf(stderr,"Q4_STATE_READY components=%zu read_bytes=%llu setup_ms=%.3f exact_initial_state=1\n",values.size(),(unsigned long long)read_bytes,(Q4Tape::ns()-start)/1e6);
 }
};inline Q4State q4_state;
